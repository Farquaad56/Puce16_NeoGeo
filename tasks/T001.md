```rust
// clock.rs — 1 tick = 1 mclk (horloge maître 24 MHz)
pub type Tick = u64;
pub const MCLK_PER_68K: u64 = 2;      // 68000 = 12 MHz
pub const MCLK_PER_Z80: u64 = 6;      // Z80   =  4 MHz
pub const MCLK_PER_YM: u64 = 3;       // YM2610 = 8 MHz
pub const MCLK_PER_PIXEL: u64 = 4;    // pixel = 6 MHz
pub const PIXELS_PER_LINE: u64 = 384;
pub const MCLK_PER_LINE: u64 = 1536;
pub const LINES_PER_FRAME: u64 = 264;
pub const MCLK_PER_FRAME: u64 = 405_504;
pub const MCLK_HZ_MVS: u64 = 24_000_000;
pub const MCLK_HZ_AES: u64 = 24_167_829;

// (pas de host.rs : les interfaces CPU viennent des crates puce-m68k et puce-z80, voir T005)

// video.rs
pub const SCREEN_W: usize = 320;
pub const SCREEN_H: usize = 224;
pub struct Framebuffer { pub pixels: Vec<u32> }   // 0x00RRGGBB, SCREEN_W*SCREEN_H entrées
impl Framebuffer { pub fn new() -> Self; pub fn crc32(&self) -> u32; pub fn write_ppm(&self, path: &std::path::Path) -> std::io::Result<()>; }
impl Default for Framebuffer { fn default() -> Self { Self::new() } }

// input.rs : bits actifs-BAS exactement comme le matériel (1 = relâché)
#[derive(Clone, Copy, PartialEq, Eq, Debug)]
pub struct PadState { pub bits: u8 }   // bit7 D, bit6 C, bit5 B, bit4 A, bit3 Droite, bit2 Gauche, bit1 Bas, bit0 Haut
impl Default for PadState { fn default() -> Self { PadState { bits: 0xFF } } }
#[derive(Clone, Copy, PartialEq, Eq, Debug)]
pub struct SystemButtons { pub bits: u8 } // bit0 Start1, bit1 Select1, bit2 Start2, bit3 Select2, bit4 Coin1, bit5 Coin2, bit6 Service, bit7 Test ; 1 = relâché
impl Default for SystemButtons { fn default() -> Self { SystemButtons { bits: 0xFF } } }
pub trait InputProvider { fn pad(&self, player: usize) -> PadState; fn system(&self) -> SystemButtons; }

// crc.rs
pub fn crc32_bytes(data: &[u8]) -> u32;   // CRC-32/ISO-HDLC (poly réfléchi 0xEDB88320, init 0xFFFFFFFF, xorout 0xFFFFFFFF)
```
`Framebuffer::crc32` = `crc32_bytes` appliqué aux pixels sérialisés en u32 little-endian (4 octets par pixel, dans l'ordre ligne par ligne). `write_ppm` : format P6 binaire 320 224 255, R=bits 16-23, G=8-15, B=0-7. La table CRC est calculée au premier appel (std::sync::OnceLock<[u32;256]>) – pas de dépendance externe.

## Tests à écrire (valeurs exactes)
- `MCLK_PER_FRAME == MCLK_PER_LINE * LINES_PER_FRAME` ; `MCLK_PER_LINE == PIXELS_PER_LINE * MCLK_PER_PIXEL`.
- `Framebuffer::new().pixels.len() == 71680` et tous les pixels valent 0.
- `crc32_bytes(b"123456789") == 0xCBF43926`.
- `Framebuffer::new().crc32() == 0xE7C953CF` (CRC de 286 720 octets nuls).
- Après `fb.pixels[0] = 0x00112233` le crc32 change ; `PadState::default().bits == 0xFF`.
- `write_ppm` dans un dossier temporaire : fichier de taille 11 + 320*224*3 octets (en-tête `P6\n320 224\n255\n` = 15 octets : la taille attendue est donc 15 + 215040 = 215055).
